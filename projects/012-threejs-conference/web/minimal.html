锁定 Three.js 0.185.1 / three-mesh-bvh 0.9.13。此天气模块目前实现降雨。TSL 材质使用 WebGPURenderer；传统 WebGLRenderer 需另做适配。

      
每帧调用顺序
import { createWeather, createWetGround,
  createDropletMaterial, createWalkControls }
  from '@research/rain-kit';

// renderer 已 init，scene 中已有模型和灯光
const ground = createWetGround({ scene, y: 0 });
const drops = createDropletMaterial({ uvChannel: 0 });
mesh.material = drops.material;
const walk = createWalkControls({ camera,
  domElement: renderer.domElement, ground: ground.mesh });
walk.setActive(true);
const weather = await createWeather({ scene, renderer, camera });

let previous = performance.now();
renderer.setAnimationLoop(now => {
  const dt = Math.min((now - previous) / 1000, .05);
  previous = now;
  walk.update(dt);
  weather.update(dt);
  ground.update(dt); drops.update(dt);
  scene.updateMatrixWorld(true);
  camera.updateMatrixWorld(true);
  ground.updateReflection(renderer, camera);
  renderer.render(scene, camera);
});

// 页面卸载：先停循环，再释放模块与宿主资源
renderer.setAnimationLoop(null);
walk.dispose(); weather.dispose();
ground.dispose(); drops.dispose();
